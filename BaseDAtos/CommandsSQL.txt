CREATE TABLE Alumno (
    Boleta VARCHAR(15)  PRIMARY KEY,
    IdUsuario INT NOT NULL,
    IDGrupo VARCHAR(5) NOT NULL,

        REFERENCES Grupo(IDGrupo),

    CONSTRAINT FK_Alumno_Carrera
        FOREIGN KEY (IdCarrera)
        REFERENCES Carrera(IdCarrera),

    CONSTRAINT UQ_Alumno_Usuario UNIQUE (Boleta)
);

CREATE TABLE Asesorias (
    IdAsesoria INT PRIMARY KEY,
    Nombre VARCHAR(100) NOT NULL,
    PeriodoEscolar VARCHAR(7) NOT NULL,
    CupoDisponible INT NOT NULL,
    FechaIni DATE NOT NULL,
    FechaFin DATE NOT NULL,
    StatusActiva BOOLEAN NOT NULL,
    StatusHorario BOOLEAN NOT NULL,
    IdMateriaImpartida INT NOT NULL,

    CONSTRAINT FK_Asesorias_MateriasImpartidas
        FOREIGN KEY (IdMateriaImpartida)
        REFERENCES MateriasImpartidas(IdMateriaImpartida),

    CONSTRAINT CHK_Asesorias_Fechas
        CHECK (FechaFin >= FechaIni)
);


CREATE TABLE Horarios (
    IdHorario INT PRIMARY KEY,
    HorarioIni TIME NOT NULL,
    HorarioFin TIME NOT NULL,
    DiasAsignados VARCHAR(20) NOT NULL,
    IdAsesoria INT NOT NULL,

    CONSTRAINT FK_Horarios_Asesorias
        FOREIGN KEY (IdAsesoria)
        REFERENCES Asesorias(IdAsesoria),

    CONSTRAINT CHK_Horarios_Hora
        CHECK (HorarioFin > HorarioIni)
);


CREATE TABLE Cuestionarios (
    IdCuestionario INT PRIMARY KEY,
    Nombre VARCHAR(100) NOT NULL,
    FechaCreacion DATE NOT NULL,
    Matricula VARCHAR(15) NOT NULL,

    CONSTRAINT FK_Cuestionarios_Asesor
        FOREIGN KEY (Matricula)
        REFERENCES Asesor(Matricula)
);



CREATE TABLE TipoPregunta (
    IdTipoPregunta INT PRIMARY KEY,
    TipoPregunta VARCHAR(50) NOT NULL,

    CONSTRAINT UQ_TipoPregunta UNIQUE (TipoPregunta)
);

CREATE TABLE Preguntas (
    IdPregunta INT PRIMARY KEY,
    NumeroPregunta INT NOT NULL,
    Pregunta VARCHAR(255) NOT NULL,
    IdTipoPregunta INT NOT NULL,
    IdCuestionario INT NOT NULL,

    CONSTRAINT FK_Preguntas_Cuestionarios
        FOREIGN KEY (IdCuestionario)
        REFERENCES Cuestionarios(IdCuestionario),

    CONSTRAINT FK_Preguntas_TipoPregunta
        FOREIGN KEY (IdTipoPregunta)
        REFERENCES TipoPregunta(IdTipoPregunta),

    CONSTRAINT UQ_Pregunta_Cuestionario
        UNIQUE (IdCuestionario, NumeroPregunta)
);



CREATE TABLE Respuestas (
    IdRespuesta INT PRIMARY KEY,
    IdPregunta INT NOT NULL,
    Respuesta VARCHAR(255) NOT NULL,

    CONSTRAINT FK_Respuestas_Preguntas
        FOREIGN KEY (IdPregunta)
        REFERENCES Preguntas(IdPregunta)
);



CREATE TABLE RespuestasCorrectas (
    IdRespuestaCorrecta INT PRIMARY KEY,
    IdPregunta INT NOT NULL,
    IdRespuesta INT NOT NULL,

    CONSTRAINT FK_RespuestasCorrectas_Preguntas
        FOREIGN KEY (IdPregunta)
        REFERENCES Preguntas(IdPregunta),

    CONSTRAINT FK_RespuestasCorrectas_Respuestas
        FOREIGN KEY (IdRespuesta)
        REFERENCES Respuestas(IdRespuesta),

    CONSTRAINT UQ_RespuestaCorrecta_Pregunta
        UNIQUE (IdPregunta, IdRespuesta)
);



CREATE TABLE AlumnosAsesoriasCuestionario (
    IdAlumnosAsesoriasCuestionario INT PRIMARY KEY,
    Boleta VARCHAR(15) NOT NULL,
    IdAsesoria INT NOT NULL,
    IdCuestionario INT NOT NULL,

    CONSTRAINT FK_AAC_Alumno
        FOREIGN KEY (Boleta)
        REFERENCES Alumno(Boleta),

    CONSTRAINT FK_AAC_Asesoria
        FOREIGN KEY (IdAsesoria)
        REFERENCES Asesorias(IdAsesoria),

    CONSTRAINT FK_AAC_Cuestionario
        FOREIGN KEY (IdCuestionario)
        REFERENCES Cuestionarios(IdCuestionario)
);


CREATE TABLE CalificacionPregunta (
    IdCalifPreg INT PRIMARY KEY,
    OpSelect VARCHAR(255) NOT NULL,
    Validacion BOOLEAN NOT NULL,
    IdAlumnosAsesoriasCuestionario INT NOT NULL,
    IdPregunta INT NOT NULL,

    CONSTRAINT FK_Calificacion_AAC
        FOREIGN KEY (IdAlumnosAsesoriasCuestionario)
        REFERENCES AlumnosAsesoriasCuestionario(IdAlumnosAsesoriasCuestionario),

    CONSTRAINT FK_Calificacion_Pregunta
        FOREIGN KEY (IdPregunta)
        REFERENCES Preguntas(IdPregunta),

    CONSTRAINT UQ_Calificacion_AlumnoCuestionario_Pregunta
        UNIQUE (IdAlumnosAsesoriasCuestionario, IdPregunta)
);



CREATE TABLE Valoración (
